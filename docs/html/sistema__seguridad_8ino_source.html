3 * @brief Sistema de seguridad embebido con FreeRTOS para Arduino Mega.
4 *
5 * @details
6 * Sistema de control de acceso y monitoreo ambiental basado en FreeRTOS.
7 * Integra los siguientes módulos de hardware:
8 * * Teclado matricial 4×4 para ingreso de clave numérica.
9 * * Pantalla LCD 16×2 para retroalimentación visual al usuario.
10 * * Servo-motor como mecanismo de cerradura electrónica.
11 * * Lector RFID MFRC522 para autenticación por tarjeta.
12 * * Sensor KY-013 (NTC) para temperatura ambiente.
13 * * Sensor KY-018 (LDR) para nivel de iluminación.
14 * * Sensor KY-037 (micrófono) para detección de ruido.
15 * * Sensor KY-035 (efecto Hall) para detección de movimiento.
16 *
17 * La lógica se modela como una Máquina de Estados Finita (FSM).
18 * Todas las esperas utilizan ticks de FreeRTOS; no se usa `millis()` ni
19 * `delay()` fuera de contextos de tarea.
20 *

323/**
324 * @brief Escribe dos líneas de texto en el LCD 16×2.
325 *
326 * @details
327 * Limpia el display y rellena cada línea con espacios hasta 16 caracteres.
328 * Si la cadena supera los 16 caracteres, se trunca.
329 *
330 * @param [in] l1 Texto para la fila 0 (máx. 16 caracteres).

337 lcd.clear();

339 while (l2.length() < 16) l2 += ' ';
340 lcd.setCursor(0, 0); lcd.print(l1.substring(0, 16));
341 lcd.setCursor(0, 1); lcd.print(l2.substring(0, 16));
342}

406/**
407 * @brief Detecta la pulsación del botón físico con anti-rebote por ticks.
408 *
409 * @details
410 * Debe invocarse una vez por tick (cada 100 ms) desde @ref manejarBloqueo().
411 * Acumula ticks con nivel LOW; retorna `true` solo al soltar el botón si
412 * estuvo presionado al menos `TICKS_DEBOUNCE_BTN` ticks consecutivos.
413 *
414 * @retval true Pulsación válida detectada.
415 * @retval false No hubo pulsación válida en este tick.
416 */
417bool botonPresionado() {
418 static int cntBtn = 0;
419 if (digitalRead(PIN_BOTON) == LOW) {
420 cntBtn++;
421 } else {
422 if (cntBtn >= TICKS_DEBOUNCE_BTN) { cntBtn = 0; return true; }
423 cntBtn = 0;
424 }
425 return false;
426}

435/**
436 * @brief Lee la temperatura con el termistor NTC del KY-013.
437 *
438 * @details
439 * Convierte la lectura ADC del pin `PIN_TEMP` a temperatura mediante la
440 * ecuación de Steinhart-Hart de tres coeficientes:
441 * @code{.cpp}
442 * float R2 = TERM_R1 * (1023.0 / Vo - 1.0);
443 * float logR2 = log(R2);
444 * float T = 1.0 / (A + B*logR2 + C*logR2^3);
445 * return T - 273.15;
446 * @endcode
447 * Si el ADC devuelve 0 (circuito abierto), retorna `-999` como centinela.
448 * La alarma se dispara cuando el valor supera el umbral @ref UMBRAL_TEMP.
449 *
450 * @par Returns
451 * Temperatura en grados Celsius, o `-999` si la lectura es inválida.
452 *
453 * @see TaskAmbiental(), UMBRAL_TEMP
454 */
455float leerTemp() {
456 int Vo = analogRead(PIN_TEMP);
457 if (Vo <= 0) return -999;
458 float R2 = TERM_R1 * (1023.0 / (float)Vo - 1.0);
459 float logR2 = log(R2);
460 float T = 1.0 / (0.001129148 + 0.000234125 * logR2
461 + 0.0000000876741 * logR2 * logR2 * logR2);
462 return T - 273.15;
463}

526/**
527 * @brief Ejecuta la transición de la FSM al estado indicado.
528 *
529 * @details
530 * Función central de la FSM. En cada transición:
531 * # Actualiza `estadoActual`.
532 * # Llama a @ref limpiarSalidas().
533 * # Reinicia `textoClave` y `ticsParpadeo`.
534 * # Ejecuta las acciones de entrada del estado destino:
535 *
536 * | Estado destino | Acción de entrada |
537 * | :----------------------- | :--------------------------------------- |
538 * | `EST_INICIO` | Muestra pantalla de bienvenida en LCD |
539 * | `EST_ABRIENDO` | Llama a @ref abrirPuerta() |
540 * | `EST_MONITOR_INTRUSOS` | `vTaskResume(hIntrusos)` |
541 * | `EST_MONITOR_AMBIENTAL` | `vTaskResume(hAmbiental)` |
542 * | `EST_CONFIG` | Muestra menú de configuración |
543 * | `EST_ALARMA` | Activa buzzer (900 Hz) y `LED_ROJO` |
544 * | `EST_BLOQUEO` | Activa buzzer (600 Hz) y parpadeo rojo |
545 *
546 * @param [in] nuevo Estado destino de la transición.
547 *
548 * @par Returns
549 * Nothing.
550 *
551 * @see activarAlarma(), manejarAlarma(), manejarBloqueo()
552 */
553void irAEstado(Estado nuevo) {
554 estadoActual = nuevo;
555 limpiarSalidas();
556 textoClave = "";
557 ticsParpadeo = 0;
558 Serial.print(">> Estado: ");
559
560 switch (nuevo) {
561 case EST_INICIO:
562 Serial.println("INICIO");
563 lcdEscribir("Acerque tarjeta", "o ingrese clave:");
564 break;
565 case EST_ABRIENDO:
566 Serial.println("ABRIENDO");
567 lcdEscribir("Acceso permitido", "Puerta abierta ");
568 abrirPuerta();
569 break;
570 case EST_MONITOR_INTRUSOS:
571 Serial.println("MONITOR INTRUSOS");
572 vTaskResume(hIntrusos);
573 break;
574 case EST_MONITOR_AMBIENTAL:
575 Serial.println("MONITOR AMBIENTAL");
576 vTaskResume(hAmbiental);
577 break;
578 case EST_CONFIG:
579 Serial.println("CONFIG");
580 lcdEscribir("CONFIG ", "1=Clave 2=RFID *");
581 break;
582 case EST_ALARMA:
583 Serial.print("ALARMA: "); Serial.println(razonAlarma);
584 ticsAlarma = 0; ticsParpadeo = 0;
585 ledOn = true;
586 digitalWrite(LED_ROJO, HIGH);
587 tone(BUZZER, 900);
588 lcdEscribir("!! ALARMA !! ", razonAlarma);
589 break;
590 case EST_BLOQUEO:
591 Serial.println("BLOQUEO");
592 ticsParpadeo = 0; ledOn = true;
593 digitalWrite(LED_ROJO, HIGH);
594 tone(BUZZER, 600);
595 lcdEscribir("SISTEMA ", "BLOQUEADO ");
596 break;
597 }
598}

603 * @details
604 * Guarda el estado de origen en `estadoAnterior` para restaurarlo al
605 * finalizar la alarma. Controla el contador de alarmas consecutivas:
606 * si la alarma ocurre dentro de `TICKS_VENTANA` ticks desde la anterior,
607 * incrementa `alarmasConsec`; de lo contrario lo reinicia a 1. Al alcanzar
608 * `MAX_ALARMAS` alarmas consecutivas, @ref manejarAlarma() transicionará
609 * a `EST_BLOQUEO` en lugar de regresar al estado anterior.
610 *
611 * @param [in] razon Cadena de hasta 16 caracteres con la causa de la alarma.
612 * @param [in] origen Estado desde el cual se dispara la alarma.
613 *
614 * @par Returns
615 * Nothing.
616 *
617 * @see irAEstado(), manejarAlarma(), estadoAnterior, alarmasConsec
618 */
619void activarAlarma(String razon, Estado origen) {
620 estadoAnterior = origen;
621 while (razon.length() < 16) razon += ' ';
622 razonAlarma = razon.substring(0, 16);
623 alarmasConsec = (ticsGlobal - ticsUltimaAlarma < TICKS_VENTANA)
624 ? alarmasConsec + 1 : 1;
625 ticsUltimaAlarma = ticsGlobal;
626 Serial.print("Alarmas consec: "); Serial.println(alarmasConsec);
627 irAEstado(EST_ALARMA);
628}

637/**
638 * @brief Lee una tarjeta RFID y autoriza o deniega el acceso.
639 *
640 * @details
641 * Construye el UID en hexadecimal (mayúsculas) y lo compara con
642 * `uidAutorizado`:
643 *
644 * * **UID válido**: reinicia `intentosFallidos`, emite beep de 1500 Hz,
645 * y llama a `irAEstado(EST_ABRIENDO)`.
646 * * **UID inválido**: incrementa `intentosFallidos`; al alcanzar 3
647 * llama a `irAEstado(EST_BLOQUEO)`.
648 *
649 * Usa `vTaskDelay` internamente, por lo que **solo debe invocarse desde
650 * una tarea FreeRTOS** (@ref TaskControl vía @ref manejarInicio()).
651 *
652 * @par Returns
653 * Nothing.
654 *
655 * @see manejarInicio(), irAEstado(), intentosFallidos
656 */
657void verificarRFID() {
658 if (!rfid.PICC_IsNewCardPresent() || !rfid.PICC_ReadCardSerial()) return;
659 String uid = "";
660 for (byte i = 0; i < rfid.uid.size; i++) {
661 if (rfid.uid.uidByte[i] < 0x10) uid += "0";
662 uid += String(rfid.uid.uidByte[i], HEX);
663 }
664 uid.toUpperCase();
665 Serial.print("RFID: "); Serial.println(uid);
666
667 if (uid == uidAutorizado) {
668 intentosFallidos = 0;
669 tone(BUZZER, 1500, 300);
670 vTaskDelay(pdMS_TO_TICKS(300));
671 estadoPost = EST_MONITOR_AMBIENTAL;
672 irAEstado(EST_ABRIENDO);
673 } else {
674 lcdEscribir("RFID INVALIDO ", "Acceso denegado ");
675 digitalWrite(LED_ROJO, HIGH);
676 tone(BUZZER, 300, 1000);
677 intentosFallidos++;
678 Serial.print("RFID invalido. Intentos: "); Serial.println(intentosFallidos);
679 vTaskDelay(pdMS_TO_TICKS(1500));
680 limpiarSalidas();
681 if (intentosFallidos >= 3) irAEstado(EST_BLOQUEO);
682 else irAEstado(EST_INICIO);
683 }
684 rfid.PICC_HaltA();
685}

751/**
752 * @brief Tarea FreeRTOS principal: ciclo de despacho de la FSM.
753 *
754 * @details
755 * Se ejecuta cada `PASO_MS` ms. En cada iteración:
756 * # Incrementa `ticsGlobal`.
757 * # Despacha al manejador del estado activo:
758 *
759 * | Estado activo | Manejador invocado |
760 * | :---------------------- | :---------------------- |
761 * | `EST_INICIO` | @ref manejarInicio() |
762 * | `EST_ABRIENDO` | @ref manejarAbriendo() |
763 * | `EST_CONFIG` | @ref manejarConfig() |
764 * | `EST_ALARMA` | @ref manejarAlarma() |
765 * | `EST_BLOQUEO` | @ref manejarBloqueo() |
766 *
767 * Los estados `EST_MONITOR_INTRUSOS` y `EST_MONITOR_AMBIENTAL` son
768 * gestionados por sus propias tareas (@ref TaskIntrusos,
769 * @ref TaskAmbiental) que se suspenden y reanudan según sea necesario.
770 *
771 * @param [in] pv Parámetro de tarea FreeRTOS (no usado; pásese `NULL`).
772 *
773 * @par Returns
774 * Nothing.
775 */
776void TaskControl(void *pv) {
777 vTaskDelay(pdMS_TO_TICKS(1500));
778 irAEstado(EST_INICIO);
779
780 for (;;) {
781 ticsGlobal++;
782 switch (estadoActual) {
783 case EST_INICIO: manejarInicio(); break;
784 case EST_ABRIENDO: manejarAbriendo(); break;
785 case EST_CONFIG: manejarConfig(); break;
786 case EST_ALARMA: manejarAlarma(); break;
787 case EST_BLOQUEO: manejarBloqueo(); break;
788 default: break;
789 }
790 vTaskDelay(pdMS_TO_TICKS(PASO_MS));
791 }
792}

794/**
795 * @brief Tarea FreeRTOS de monitoreo ambiental (temperatura y luz).
796 *
797 * @details
798 * Se suspende al crearse y se reanuda desde @ref irAEstado() cuando el
799 * estado destino es `EST_MONITOR_AMBIENTAL`. En cada ciclo de
800 * `CICLOS_AMBIENTAL` ticks (5 s):
801 *
802 * * Lee @ref leerTemp() y @ref leerLuz() cada tick.
803 * * Actualiza el LCD cada 10 ticks con valores en tiempo real.
804 * * `temp < UMBRAL_TEMP` → llama a @ref activarAlarma() con `"TEMP BAJA <20C"`.
805 * * `luz < UMBRAL_LUZ` → llama a @ref activarAlarma() con `"LUZ BAJA"`.
806 * * Tecla `'*'` → transiciona a `EST_CONFIG`.
807 *
808 * Al finalizar el ciclo sin incidencias, cede a `EST_MONITOR_INTRUSOS`
809 * y se suspende de nuevo.
810 *
811 * @param [in] pv Parámetro de tarea FreeRTOS (no usado; pásese `NULL`).
812 *
813 * @par Returns
814 * Nothing.
815 *
816 * @see leerTemp(), leerLuz(), activarAlarma(), TaskIntrusos()
817 */
818void TaskAmbiental(void *pv) {
819 vTaskSuspend(NULL);
820 const TickType_t paso = pdMS_TO_TICKS(PASO_MS);
821
822 for (;;) {
823 limpiarSalidas();
824 estadoActual = EST_MONITOR_AMBIENTAL;
825 Serial.println(">> MONITOR AMBIENTAL");
826 lcdEscribir("[AMBIENT] OK ", "T:--C L:--- ");
827
828 for (int c = 0; c < CICLOS_AMBIENTAL; c++) {
829 float temp = leerTemp();
830 int luz = leerLuz();
831 if (c % 10 == 0) {
832 String l2 = "T:" + String((int)temp) + "C L:" + String(luz) + " ";
833 lcdEscribir("[AMBIENT] OK ", l2);
834 }
835 if (temp < UMBRAL_TEMP) { activarAlarma("TEMP BAJA <20C ", EST_MONITOR_AMBIENTAL); break; }
836 if (luz < UMBRAL_LUZ ) { activarAlarma("LUZ BAJA ", EST_MONITOR_AMBIENTAL); break; }
837 char k = keypad.getKey();
838 if (k == '*') { estadoAnterior = EST_MONITOR_AMBIENTAL; irAEstado(EST_CONFIG); break; }
839 vTaskDelay(paso);
840 }
841 if (estadoActual == EST_MONITOR_AMBIENTAL) irAEstado(EST_MONITOR_INTRUSOS);
842 vTaskSuspend(NULL);
843 }
844}

846/**
847 * @brief Tarea FreeRTOS de monitoreo de intrusos.
848 *
849 * @details
850 * Se suspende al crearse y se reanuda desde @ref irAEstado() cuando el
851 * estado destino es `EST_MONITOR_INTRUSOS`. En cada ciclo de
852 * `CICLOS_INTRUSOS` ticks (2 s) evalúa tres fuentes de amenaza:
853 *
854 * **1. Puerta abierta** (@ref puertaAb())
855 * Tres ticks consecutivos positivos disparan `"PUERTA ABIERTA"`.
856 *
857 * **2. Campo magnético** (@ref hallDet())
858 * Tres ticks consecutivos positivos disparan `"MOVIMIENTO DET."`.
859 *
860 * **3. Micrófono** (@ref leerMic()) — lógica de eventos acumulados:
861 * * `MIC_LECTURAS_EVENTO` ticks con ruido → evento confirmado.
862 * * Evento termina tras `MIC_TICKS_SILENCIO` ticks sin ruido.
863 * * `MIC_EVENTOS_ALARMA` eventos → alarma `"RUIDO DETECTADO"`.
864 * * `MIC_TICKS_VENTANA` ticks sin evento → contador reiniciado.
865 *
866 * Tecla `'#'` transiciona a `EST_CONFIG`. Al finalizar el ciclo sin
867 * incidencias, cede a `EST_MONITOR_AMBIENTAL` y se suspende de nuevo.
868 *
869 * @param [in] pv Parámetro de tarea FreeRTOS (no usado; pásese `NULL`).
870 *
871 * @par Returns
872 * Nothing.
873 *
874 * @see puertaAb(), hallDet(), leerMic(), activarAlarma(), TaskAmbiental()
875 */
876void TaskIntrusos(void *pv) {
877 vTaskSuspend(NULL);
878 const TickType_t paso = pdMS_TO_TICKS(PASO_MS);
879
880 static int cntPta = 0, cntHall = 0;
881 static int micLecturasConsec = 0, micEventos = 0;
882 static bool micEnEvento = false;
883 static int micTicksSilencio = 0, micTicksVentana = 0;
884
885 for (;;) {
886 limpiarSalidas();
887 estadoActual = EST_MONITOR_INTRUSOS;
888 Serial.println(">> MONITOR INTRUSOS");
889 lcdEscribir("[INTRUSOS] OK ", "PTA:cerr mic:ok ");
890
891 for (int c = 0; c < CICLOS_INTRUSOS; c++) {
892 bool ruido = leerMic() > UMBRAL_MIC;
893
894 if (ruido) {
895 micTicksSilencio = 0; micTicksVentana = 0;
896 if (!micEnEvento) {
897 micLecturasConsec++;
898 if (micLecturasConsec >= MIC_LECTURAS_EVENTO) {
899 micLecturasConsec = 0; micEnEvento = true; micEventos++;
900 Serial.print("MIC evento #"); Serial.println(micEventos);
901 String l2 = "mic:RUIDO #"; l2 += String(micEventos); l2 += "/3 ";
902 lcdEscribir("[INTRUSOS] OK ", l2);
903 if (micEventos >= MIC_EVENTOS_ALARMA) {
904 micEventos=0; micLecturasConsec=0; micEnEvento=false;
905 micTicksSilencio=0; micTicksVentana=0;
906 activarAlarma("RUIDO DETECTADO ", EST_MONITOR_INTRUSOS); break;
907 }
908 }
909 }
910 } else {
911 micLecturasConsec = 0;
912 if (micEnEvento) {
913 if (++micTicksSilencio >= MIC_TICKS_SILENCIO) { micEnEvento=false; micTicksSilencio=0; }
914 } else {
915 if (++micTicksVentana >= MIC_TICKS_VENTANA) {
916 micEventos=0; micTicksVentana=0;
917 Serial.println("MIC ventana expirada: reset eventos");
918 }
919 }
920 }
921
922 if (c % 8 == 0) {
923 String l2 = puertaAb() ? "PTA:ABIERTA " : "PTA:cerr ";
924 l2 += ruido ? "MIC:!" : "mic:ok";
925 if (!ruido && !micEnEvento) lcdEscribir("[INTRUSOS] OK ", l2);
926 }
927 if (puertaAb()) { if (++cntPta >= 3) { cntPta=0; activarAlarma("PUERTA ABIERTA ", EST_MONITOR_INTRUSOS); break; } } else cntPta=0;
928 if (hallDet()) { if (++cntHall >= 3) { cntHall=0; activarAlarma("MOVIMIENTO DET. ", EST_MONITOR_INTRUSOS); break; } } else cntHall=0;
929 char k = keypad.getKey();
930 if (k == '#') { estadoAnterior = EST_MONITOR_AMBIENTAL; irAEstado(EST_CONFIG); break; }
931 vTaskDelay(paso);
932 }
933 if (estadoActual == EST_MONITOR_INTRUSOS) irAEstado(EST_MONITOR_AMBIENTAL);
934 vTaskSuspend(NULL);
935 }
936}

1020/**
1021 * @brief Manejador del estado `EST_ALARMA`.
1022 *
1023 * @details
1024 * Gestiona el parpadeo del LED rojo y la duración total de la alarma
1025 * usando únicamente contadores de ticks:
1026 * * LED encendido `TICKS_PARP_ON` ticks → apagado (noTone).
1027 * * LED apagado `TICKS_PARP_OFF` ticks → encendido (tone 900 Hz).
1028 *
1029 * Al cumplirse `TICKS_ALARMA` ticks totales, limpia salidas y:
1030 * * `alarmasConsec >= MAX_ALARMAS` → @ref irAEstado() a `EST_BLOQUEO`.
1031 * * En caso contrario → @ref irAEstado() a `estadoAnterior`.
1032 *
1033 * @par Returns
1034 * Nothing.
1035 *
1036 * @see activarAlarma(), irAEstado(), alarmasConsec, estadoAnterior

1038void manejarAlarma() {

1041 if (!ledOn && ticsParpadeo >= TICKS_PARP_OFF) { ledOn=true; ticsParpadeo=0; digitalWrite(LED_ROJO,HIGH); tone(BUZZER,900); }
1042 if (ticsAlarma >= TICKS_ALARMA) {

1045 irAEstado(alarmasConsec >= MAX_ALARMAS ? EST_BLOQUEO : estadoAnterior);
1046 }

1101 } else {

1105 lcdEscribir("Clave admin: ", ast);
1106 }
1107}

1109/**
1110 * @brief Manejador del estado `EST_CONFIG`.
1111 *
1112 * @details
1113 * Gestiona un submenú de dos opciones usando la variable estática `submenu`:
1114 *
1115 * | `submenu` | Descripción | Confirmación |
1116 * | :-------: | :--------------------------------- | :----------- |
1117 * | 0 | Menú principal (teclas 1, 2, \*) | — |
1118 * | 1 | Cambio de clave de usuario | `'#'` |
1119 * | 2 | Registro de nueva tarjeta RFID | Lectura RFID |
1120 *
1121 * **Submenú 1 (cambio de clave):**
1122 * Acumula dígitos; `'#'` guarda si hay ≥4 dígitos en `claveCorrecta`;
1123 * `'*'` cancela.
1124 *
1125 * **Submenú 2 (registro RFID):**
1126 * Espera lectura del MFRC522 y guarda el UID en `uidAutorizado`;
1127 * `'*'` cancela.
1128 *
1129 * La tecla `'*'` en el menú principal llama a
1130 * @ref irAEstado() con `estadoAnterior`.
1131 *
1132 * @par Returns
1133 * Nothing.
1134 *
1135 * @see irAEstado(), claveCorrecta, uidAutorizado
1136 */
1137void manejarConfig() {
1138 static int submenu = 0;
1139 char key = keypad.getKey();
1140
1141 if (submenu == 1) {
1142 if (!key) return;
1143 if (key == '#') {
1144 if (textoClave.length() >= 4) {
1145 claveCorrecta = textoClave;
1146 lcdEscribir("Clave guardada ", textoClave);
1147 Serial.print("Nueva clave: "); Serial.println(claveCorrecta);
1148 vTaskDelay(pdMS_TO_TICKS(1500));
1149 } else {
1150 lcdEscribir("Min 4 digitos ", " ");
1151 vTaskDelay(pdMS_TO_TICKS(1200));
1152 }
1153 textoClave = ""; submenu = 0;
1154 lcdEscribir("CONFIG ", "1=Clave 2=RFID *");
1155 } else if (key == '*') {
1156 textoClave = ""; submenu = 0;
